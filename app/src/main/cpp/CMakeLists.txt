cmake_minimum_required(VERSION 3.22)
project(dialtone_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
    message(FATAL_ERROR "Missing llama.cpp source. Run tools/setup_engine.sh or GitHub Actions build.")
endif()
add_subdirectory(llama.cpp ${CMAKE_CURRENT_BINARY_DIR}/llama)
add_library(dialtone-llama SHARED native-llama.cpp)
target_include_directories(dialtone-llama PRIVATE llama.cpp/include)
target_link_libraries(dialtone-llama PRIVATE llama log)
