cmake_minimum_required(VERSION 3.22)
project(dialtone_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
if(ANDROID)
    # Same setup as llama.cpp's own Android example: ship one CPU backend per ARM generation and
    # load the best one the phone supports at runtime (dotprod, int8 matmul, SVE...). Old phones still work.
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
else()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
    message(FATAL_ERROR "Missing llama.cpp source. Run tools/setup_engine.sh or GitHub Actions build.")
endif()
add_subdirectory(llama.cpp ${CMAKE_CURRENT_BINARY_DIR}/llama)

add_library(dialtone-engine STATIC dialtone-engine.cpp)
target_link_libraries(dialtone-engine PUBLIC llama)

if(ANDROID)
    add_library(dialtone-llama SHARED native-llama.cpp)
    target_link_libraries(dialtone-llama PRIVATE dialtone-engine log)
else()
    # Desktop harness for testing the caller's dialogue without a phone.
    add_executable(dialtone-chat-test chat-test.cpp)
    target_link_libraries(dialtone-chat-test PRIVATE dialtone-engine)
endif()
